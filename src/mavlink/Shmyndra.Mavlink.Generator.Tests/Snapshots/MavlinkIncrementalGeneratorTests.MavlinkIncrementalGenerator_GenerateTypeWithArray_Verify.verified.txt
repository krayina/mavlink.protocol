namespace MavlinkTypes.Testfile
{
    /// <remarks>
    /// Original name: TEST_MESSAGE
    /// </remarks>
    [MavlinkIdentifiedType(0U, "TEST_MESSAGE")]
    public record struct TestMessage : MavlinkMessage
    {
        /// <remarks>
        /// Original name: URI
        /// </remarks>
        [System.ComponentModel.DataAnnotations.RequiredArrayLength(230)]
        public System.Collections.Immutable.ImmutableArray<char> Uri { get; init; }
    }

    public static class MavlinkSpecification
    {
        public static byte? Version = null;
        public static byte? Dialect = null;
    }
}
namespace MavlinkTypes
{
    public static class MavlinkMessages
    {
        private static Dictionary<Type, (ulong Id, string XmlName)> _mavlinkMessages = new()
        {
            {
                typeof(MavlinkTypes.Testfile.TestMessage),
                (0U, "TEST_MESSAGE")
            }
        };
        public static ulong GetId<T>()
            where T : MavlinkMessage
        {
            return _mavlinkMessages[typeof(T)].Id;
        }

        public static string GetXmlName<T>()
            where T : MavlinkMessage
        {
            return _mavlinkMessages[typeof(T)].XmlName;
        }
    }
}

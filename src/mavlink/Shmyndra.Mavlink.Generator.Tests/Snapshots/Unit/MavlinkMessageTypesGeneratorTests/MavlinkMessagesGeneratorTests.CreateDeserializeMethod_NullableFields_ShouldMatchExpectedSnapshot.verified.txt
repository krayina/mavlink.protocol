public static SomeMessage Deserialize(byte[] payload)
{
    if (payload.Length == 0)
    {
        return new SomeMessage();
    }

    if (payload.Length < 0)
    {
        var paddedPayload = new byte[0];
        Array.Copy(payload, paddedPayload, payload.Length);
        payload = paddedPayload;
    }

    if (payload.Length > 0 && payload.Length < 1)
    {
        var paddedPayload = new byte[1];
        Array.Copy(payload, paddedPayload, payload.Length);
        payload = paddedPayload;
    }

    byte? index = null;
    if (payload.Length > 0)
    {
        index = payload[0];
    }

    if (payload.Length > 1 && payload.Length < 9)
    {
        var paddedPayload = new byte[9];
        Array.Copy(payload, paddedPayload, payload.Length);
        payload = paddedPayload;
    }

    ulong? timeUsec = null;
    if (payload.Length > 1)
    {
        timeUsec = BitConverter.ToUInt64(payload, 1);
    }

    if (payload.Length > 9 && payload.Length < 11)
    {
        var paddedPayload = new byte[11];
        Array.Copy(payload, paddedPayload, payload.Length);
        payload = paddedPayload;
    }

    ushort? counter = null;
    if (payload.Length > 9)
    {
        counter = BitConverter.ToUInt16(payload, 9);
    }

    if (payload.Length > 11 && payload.Length < 12)
    {
        var paddedPayload = new byte[12];
        Array.Copy(payload, paddedPayload, payload.Length);
        payload = paddedPayload;
    }

    byte? count = null;
    if (payload.Length > 11)
    {
        count = payload[11];
    }

    if (payload.Length > 12 && payload.Length < 13)
    {
        var paddedPayload = new byte[13];
        Array.Copy(payload, paddedPayload, payload.Length);
        payload = paddedPayload;
    }

    byte? connectionTypeValue = null;
    if (payload.Length > 12)
    {
        connectionTypeValue = (byte)payload[12];
    }

    var connectionType = connectionTypeValue.HasValue ? (EscConnectionType?)connectionTypeValue.Value : null;
    if (payload.Length > 13 && payload.Length < 14)
    {
        var paddedPayload = new byte[14];
        Array.Copy(payload, paddedPayload, payload.Length);
        payload = paddedPayload;
    }

    byte? info = null;
    if (payload.Length > 13)
    {
        info = payload[13];
    }

    if (payload.Length > 14 && payload.Length < 22)
    {
        var paddedPayload = new byte[22];
        Array.Copy(payload, paddedPayload, payload.Length);
        payload = paddedPayload;
    }

    System.Collections.Immutable.ImmutableArray<EscFailureFlags>? failureFlags = null;
    if (payload.Length >= 22)
    {
        var tempFailureFlagsArray = new EscFailureFlags[4];
        Buffer.BlockCopy(payload, 14, tempFailureFlagsArray, 0, 8);
        failureFlags = System.Collections.Immutable.ImmutableArray.CreateRange(tempFailureFlagsArray);
    }

    if (payload.Length > 22 && payload.Length < 38)
    {
        var paddedPayload = new byte[38];
        Array.Copy(payload, paddedPayload, payload.Length);
        payload = paddedPayload;
    }

    System.Collections.Immutable.ImmutableArray<uint>? errorCount = null;
    if (payload.Length >= 38)
    {
        var tempErrorCountArray = new uint[4];
        Buffer.BlockCopy(payload, 22, tempErrorCountArray, 0, 16);
        errorCount = System.Collections.Immutable.ImmutableArray.CreateRange(tempErrorCountArray);
    }

    if (payload.Length > 38 && payload.Length < 46)
    {
        var paddedPayload = new byte[46];
        Array.Copy(payload, paddedPayload, payload.Length);
        payload = paddedPayload;
    }

    System.Collections.Immutable.ImmutableArray<short>? temperature = null;
    if (payload.Length >= 46)
    {
        var tempTemperatureArray = new short[4];
        Buffer.BlockCopy(payload, 38, tempTemperatureArray, 0, 8);
        temperature = System.Collections.Immutable.ImmutableArray.CreateRange(tempTemperatureArray);
    }

    return new SomeMessage
    {
        Index = index,
        TimeUsec = timeUsec,
        Counter = counter,
        Count = count,
        ConnectionType = connectionType,
        Info = info,
        FailureFlags = failureFlags,
        ErrorCount = errorCount,
        Temperature = temperature
    };
}
